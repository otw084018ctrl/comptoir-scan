function startCamera() { showCenter('warn', '正在啟動相機…', '請允許瀏覽器使用相機權限'); html5QrCode = new Html5Qrcode("reader"); // 1. 拿掉 qrbox，讓它掃描「全畫面」；提升 FPS 增加抓到清晰瞬間的機率 var config = { fps: 15, experimentalFeatures: { useBarCodeDetectorIfSupported: true // 保持硬體晶片加速 } }; // 2. 強制要求瀏覽器開啟「高畫質 (FHD)」與「連續自動對焦」 var cameraConstraints = { facingMode: "environment", width: { ideal: 1920, min: 1280 }, // 強制要求至少 720p，理想是 1080p height: { ideal: 1080, min: 720 }, advanced: [{ focusMode: "continuous" }] // 呼叫底層的連續對焦馬達 }; html5QrCode.start( cameraConstraints, config, function(decodedText, decodedResult) { onHit(decodedText); }, function(errorMessage) { // 沒掃到時會持續觸發，忽略即可 } ).then(function() { hideCenter(); }).catch(function(err) { showCenter('warn', '無法開啟相機', '請確認權限，或改用下方「手動輸入」。', '改用手動輸入', function () { hideCenter(); $('manualRow').classList.add('show'); $('manualToggle').classList.add('on'); $('manualInput').focus(); }); }); }
